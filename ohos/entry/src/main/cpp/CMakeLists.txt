cmake_minimum_required(VERSION 3.16)
project(mfp_recording LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_SKIP_RPATH TRUE)
add_library(mfp_recording SHARED recording_store.cpp recording_napi.cpp)
target_compile_definitions(mfp_recording PRIVATE _GNU_SOURCE=1)
target_compile_options(mfp_recording PRIVATE -Wall -Wextra -Werror)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  target_compile_options(mfp_recording PRIVATE -Wno-error=unused-command-line-argument)
endif()
target_link_libraries(mfp_recording PRIVATE libace_napi.z.so)
